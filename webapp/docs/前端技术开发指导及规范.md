## 三、代码规范

### 3.1 命名规范

| 类型 | 规范 | 示例 |
|------|------|------|
| **组件名** | PascalCase | `UserTable.tsx`, `DomainForm.tsx` |
| **文件名（组件）** | PascalCase | `RichTextEditor.tsx` |
| **文件名（非组件）** | camelCase | `authStore.ts`, `useDebounce.ts` |
| **目录名** | kebab-case | `sensitive-words/`, `card-generator/` |
| **变量/函数** | camelCase | `getUserList()`, `pageSize` |
| **常量** | UPPER_SNAKE_CASE | `MAX_BANNER_COUNT = 5` |
| **类型/接口** | PascalCase | `DomainData`, `ApiResponse<T>` |
| **枚举** | PascalCase（枚举名）+ UPPER_SNAKE_CASE（值） | `MatchMode.Exact` |
| **React Hooks** | camelCase, use 前缀 | `useAuth`, `useTheme` |
| **CSS 类名** | kebab-case（Tailwind）或 BEM | `card-preview`, `sidebar__nav` |
| **事件处理** | handle 前缀 | `handleSubmit`, `handleDelete` |
| **布尔变量** | is/has/should 前缀 | `isLoading`, `hasError`, `shouldShow` |

## 四、组件开发规范

### 4.1 组件分类

| 分类 | 说明 | 目录 | 示例 |
|------|------|------|------|
| **页面组件** | 对应一个路由页面 | `src/pages/*/` | `DomainsPage`, `CardsPage` |
| **业务组件** | 可复用的业务逻辑模块 | `src/components/*/` | `RichTextEditor`, `ImageUploader` |
| **通用组件** | 跨业务场景的 UI 元素 | `src/components/common/` | `ConfirmDelete`, `StatusBadge` |
| **布局组件** | 页面框架结构 | `src/layouts/` | `AdminLayout`, `Sidebar` |

### 4.2 组件设计原则

```
1. 单一职责：每个组件只做一件事
2. 组合优于继承：通过组合 props 和 children 扩展
3. 状态提升：共享状态提升到最近的公共父组件
4. 接口显式：所有 props 显式定义类型，避免隐式依赖
5. 纯组件优先：尽可能做成纯展示组件，容器组件处理数据
```

### 4.3 页面组件模式

**表单使用规范**：
```tsx
// ✅ 推荐：React Hook Form + Ant Design + Zod 校验
import { useForm, Controller } from 'react-hook-form'
import { zodResolver } from '@hookform/resolvers/zod'
import { z } from 'zod'

const domainSchema = z.object({
  name: z.string().min(1, '请输入领域名称').max(20, '最长20个字符'),
  isFree: z.boolean(),
  unlockPoints: z.number().min(30).max(60).optional(),
  icon: z.string().min(1, '请选择图标'),
})

type DomainFormValues = z.infer<typeof domainSchema>

function DomainForm({ defaultValues, onSubmit }: DomainFormProps) {
  const { control, handleSubmit, watch, formState: { errors } } = useForm<DomainFormValues>({
    resolver: zodResolver(domainSchema),
    defaultValues,
  })

  const isFree = watch('isFree')

  return (
    <Form layout="vertical" onFinish={handleSubmit(onSubmit)}>
      <Form.Item label="领域名称" validateStatus={errors.name ? 'error' : ''} help={errors.name?.message}>
        <Controller
          name="name"
          control={control}
          render={({ field }) => <Input {...field} placeholder="如：编程基础概念" maxLength={20} />}
        />
      </Form.Item>

      <Form.Item label="是否免费">
        <Controller
          name="isFree"
          control={control}
          render={({ field }) => (
            <Switch checked={field.value} onChange={field.onChange} />
          )}
        />
      </Form.Item>

      {!isFree && (
        <Form.Item label="解锁积分" validateStatus={errors.unlockPoints ? 'error' : ''} help={errors.unlockPoints?.message}>
          <Controller
            name="unlockPoints"
            control={control}
            render={({ field }) => <InputNumber {...field} min={30} max={60} />}
          />
        </Form.Item>
      )}
    </Form>
  )
}
```

### 5.3 Zustand Store 规范

```tsx
// stores/authStore.ts
import { create } from 'zustand'
import { persist } from 'zustand/middleware'

interface AuthState {
  token: string | null
  username: string | null
  isAuthenticated: boolean
  login: (username: string, password: string) => Promise<void>
  logout: () => void
}

export const useAuthStore = create<AuthState>()(
  persist(
    (set) => ({
      token: null,
      username: null,
      isAuthenticated: false,

      login: async (username, password) => {
        const { token } = await authApi.login({ username, password })
        set({ token, username, isAuthenticated: true })
      },

      logout: () => {
        set({ token: null, username: null, isAuthenticated: false })
      },
    }),
    {
      name: 'codesail-auth',
      partialize: (state) => ({ token: state.token, username: state.username }),
    }
  )
)

// stores/themeStore.ts
interface ThemeState {
  mode: 'light' | 'dark'
  toggle: () => void
}

export const useThemeStore = create<ThemeState>()(
  persist(
    (set) => ({
      mode: 'light',
      toggle: () => set((state) => ({ mode: state.mode === 'light' ? 'dark' : 'light' })),
    }),
    { name: 'codesail-theme' }
  )
)
```

## 六、API 集成规范

### 6.1 Axios 实例配置

```tsx
// api/client.ts
import axios from 'axios'
import { useAuthStore } from '@/stores/authStore'

const apiClient = axios.create({
  baseURL: import.meta.env.VITE_API_BASE_URL || '/api',
  timeout: 15000,
  headers: { 'Content-Type': 'application/json' },
})

// 请求拦截器：注入 Token
apiClient.interceptors.request.use((config) => {
  const token = useAuthStore.getState().token
  if (token) {
    config.headers.Authorization = `Bearer ${token}`
  }
  return config
})

// 响应拦截器：统一错误处理
apiClient.interceptors.response.use(
  (response) => response.data,
  (error) => {
    if (error.response?.status === 401) {
      useAuthStore.getState().logout()
      window.location.href = '/login'
      return Promise.reject(new Error('登录已过期，请重新登录'))
    }
    const message = error.response?.data?.message || error.message || '请求失败'
    return Promise.reject(new Error(message))
  }
)

export default apiClient
```

### 6.2 API 模块规范

```tsx
// api/domains.ts
import apiClient from './client'
import type { DomainData, DomainFormData, PaginatedResponse, ApiResponse } from '@/types'

export const domainApi = {
  getList(params?: { page?: number; pageSize?: number; keyword?: string }) {
    return apiClient.get<ApiResponse<PaginatedResponse<DomainData>>>('/domains', { params })
  },

  getById(id: string) {
    return apiClient.get<ApiResponse<DomainData>>(`/domains/${id}`)
  },

  create(data: DomainFormData) {
    return apiClient.post<ApiResponse<DomainData>>('/domains', data)
  },

  update(id: string, data: Partial<DomainFormData>) {
    return apiClient.put<ApiResponse<DomainData>>(`/domains/${id}`, data)
  },

  delete(id: string) {
    return apiClient.delete<ApiResponse<void>>(`/domains/${id}`)
  },

  updateSort(items: { id: string; sortOrder: number }[]) {
    return apiClient.put<ApiResponse<void>>('/domains/sort', { items })
  },
}
```

## 九、AI 工作台开发规范

### 9.1 AI 流式输出

```tsx
// hooks/useAIStream.ts
import { useState, useCallback, useRef } from 'react'

interface UseAIStreamOptions {
  onToken?: (token: string) => void
  onDone?: (fullText: string) => void
  onError?: (error: Error) => void
}

export function useAIStream(options: UseAIStreamOptions = {}) {
  const [isStreaming, setIsStreaming] = useState(false)
  const [content, setContent] = useState('')
  const abortRef = useRef<AbortController | null>(null)

  const startStream = useCallback(async (url: string, body: object) => {
    const controller = new AbortController()
    abortRef.current = controller
    setIsStreaming(true)
    setContent('')

    try {
      const response = await fetch(url, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(body),
        signal: controller.signal,
      })

      if (!response.ok) throw new Error('AI 生成请求失败')

      const reader = response.body?.getReader()
      if (!reader) throw new Error('无法读取响应流')

      const decoder = new TextDecoder()
      let fullText = ''

      while (true) {
        const { done, value } = await reader.read()
        if (done) break

        const chunk = decoder.decode(value, { stream: true })
        fullText += chunk
        setContent(fullText)
        options.onToken?.(chunk)
      }

      options.onDone?.(fullText)
    } catch (error) {
      if (error instanceof Error && error.name !== 'AbortError') {
        options.onError?.(error)
      }
    } finally {
      setIsStreaming(false)
      abortRef.current = null
    }
  }, [options])

  const cancelStream = useCallback(() => {
    abortRef.current?.abort()
  }, [])

  return { isStreaming, content, startStream, cancelStream }
}
```
